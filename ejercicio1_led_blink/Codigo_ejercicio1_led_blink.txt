;=========================================================
; Código en Assembler para PIC18F4550
; V 1.0
; Parpadeo de LED en RB0:
;   - 1 segundo ENCENDIDO
;   - 2 segundos APAGADO
;
; Frecuencia: 8 MHz (Oscilador Interno)
; Ensamblador: MPLAB XC8 3.0
;
; Comentario de versión:
; Aquí añadimos el loop maestro. Esto lo hacemos con el fin de
; poder alcanzar retardos del orden de segundos con contadores 
; de 8 bits. Con este tercer loop podemos hacer una mejor 
; aproximación al segundo que nos servira despues en los 
; siguientes puntos del proyecto.
;
; === CÁLCULO DE RETARDOS ===
; En PIC18 cada instrucción toma 4 ciclos de reloj:
;   T_ciclo = 4 / 8,000,000 Hz = 0.5 µs por ciclo de instrucción
;
; Se usa una estructura de 3 bucles anidados:
;   Maestro × Externo × Interno
;
; Fórmula aproximada:
;   t ≈ CNT_MASTER × CNT_EXT × CNT_INT × 6 × 0.5 µs
;
; El factor 6 proviene del cuerpo del bucle interno
; (NOPs + DECFSZ + GOTO).
;
; --- Retardo 1 segundo ---
;   10 × 200 × 166 × 6 × 0.5 µs = 996,000 µs ≈ 1.0 s
;
; --- Retardo 2 segundos ---
;   20 × 200 × 166 × 6 × 0.5 µs = 1,992,000 µs ≈ 2.0 s
;
; El pequeño error restante se compensa con el overhead
; natural de los bucles. Estos valores se refinan después.
;=========================================================

    #include <xc.inc>

    ;=======================================================
    ; Configuración de Fuses
    ;=======================================================
    CONFIG  FOSC = INTOSCIO_EC   ; Oscilador interno a 8 MHz
    CONFIG  WDT  = OFF           ; Watchdog deshabilitado
    CONFIG  LVP  = OFF           ; Sin programación en bajo voltaje
    CONFIG  PBADEN = OFF         ; PORTB digital desde reset

    ;=======================================================
    ; Constantes simbólicas
    ;=======================================================
    #define LED_PIN   0          ; LED conectado en RB0

    ; Contadores comunes a ambos retardos
    ; Se mantienen iguales para que el ajuste fino
    ; dependa solo del contador maestro
    #define CNT_EXT   200        ; Bucle externo
    #define CNT_INT   166        ; Bucle interno (define el ciclo base)

    ; Contador maestro
    ; Introducido en esta versión para ampliar el rango
    ; temporal más allá de lo posible con solo dos bucles
    #define CNT_M_1S  10         ; ≈ 1 segundo
    #define CNT_M_2S  20         ; ≈ 2 segundos

    ;=======================================================
    ; Vector de Reset
    ;=======================================================
    PSECT  resetVec, class=CODE, reloc=2
    ORG     0x00
    GOTO    Inicio

    ;=======================================================
    ; Vectores de interrupción (no usados)
    ;=======================================================
    PSECT  highIntVec, class=CODE, reloc=2
    ORG     0x08
    RETFIE

    PSECT  lowIntVec, class=CODE, reloc=2
    ORG     0x18
    RETFIE

    ;=======================================================
    ; Código Principal
    ;=======================================================
    PSECT  main_code, class=CODE, reloc=2

Inicio:
    CLRF    TRISB                ; PORTB como salida
    CLRF    LATB                 ; LED inicialmente apagado

    ;-------------------------------------------------------
    ; Bucle principal:
    ; LED encendido 1 s → apagado 2 s → repetir
    ;-------------------------------------------------------
Loop:
    BSF     LATB, LED_PIN        ; Encender LED
    CALL    Retardo_1s           ; Retardo aproximado de 1 segundo

    BCF     LATB, LED_PIN        ; Apagar LED
    CALL    Retardo_2s           ; Retardo aproximado de 2 segundos

    GOTO    Loop                 ; Ejecución continua

    ;=======================================================
    ; Subrutina: Retardo de 1 Segundo
    ; Aquí se usa por primera vez el loop maestro.
    ; Su función es multiplicar el tiempo alcanzable
    ; sin cambiar la estructura interna del retardo.
    ;=======================================================
Retardo_1s:
    MOVFF   WREG, W_Backup       ; Preservar W (buena práctica)

    MOVLW   CNT_M_1S
    MOVWF   ContadorMaestro     ; Inicializar bucle maestro

Loop1s_Maestro:
    MOVLW   CNT_EXT
    MOVWF   ContadorExterno     ; Bucle de nivel medio

Loop1s_Externo:
    MOVLW   CNT_INT
    MOVWF   ContadorInterno     ; Bucle interno (ciclo base)

Loop1s_Interno:
    NOP                          ; Ajuste fino de tiempo
    NOP
    NOP
    DECFSZ  ContadorInterno, F
    GOTO    Loop1s_Interno

    DECFSZ  ContadorExterno, F
    GOTO    Loop1s_Externo

    DECFSZ  ContadorMaestro, F
    GOTO    Loop1s_Maestro

    MOVFF   W_Backup, WREG       ; Restaurar W
    RETURN

    ;=======================================================
    ; Subrutina: Retardo de 2 Segundos
    ; Misma estructura que el retardo de 1 s.
    ; Solo cambia el contador maestro, lo que
    ; facilita el escalado del tiempo.
    ;=======================================================
Retardo_2s:
    MOVFF   WREG, W_Backup

    MOVLW   CNT_M_2S
    MOVWF   ContadorMaestro

Loop2s_Maestro:
    MOVLW   CNT_EXT
    MOVWF   ContadorExterno

Loop2s_Externo:
    MOVLW   CNT_INT
    MOVWF   ContadorInterno

Loop2s_Interno:
    NOP
    NOP
    NOP
    DECFSZ  ContadorInterno, F
    GOTO    Loop2s_Interno

    DECFSZ  ContadorExterno, F
    GOTO    Loop2s_Externo

    DECFSZ  ContadorMaestro, F
    GOTO    Loop2s_Maestro

    MOVFF   W_Backup, WREG
    RETURN

    ;=======================================================
    ; Variables en RAM
    ;=======================================================
    PSECT udata
ContadorMaestro:    DS 1         ; Nivel superior del retardo
ContadorExterno:    DS 1         ; Nivel intermedio
ContadorInterno:    DS 1         ; Nivel base
W_Backup:           DS 1         ; Respaldo de WREG

    END
